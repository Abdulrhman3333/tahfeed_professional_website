# 2. سحب أحدث التحديثات من GitHub (تأكد من اسم الفرع main أو master)
git pull origin main

# 3. تفعيل بيئة العمل الافتراضية (Virtual Environment)
source ~/.virtualenvs/اسم_بيئة_العمل/bin/activate

# 4. تثبيت أي مكتبات أو حزم جديدة تم إضافتها للمشروع (اختياري)
pip install -r requirements.txt

# 5. تطبيق التعديلات على قاعدة البيانات
python manage.py migrate

# 6. تجميع وتحديث الملفات الثابتة مثل CSS و JS والصور
python manage.py collectstatic --noinput
